  29  def mask_to_geometry(mask: np.ndarray, mm_per_pixel: float, simplify_mm: float):
  30      """Polygons whose edges follow pixel edges (not pixel centres)."""
  31  
  32      contours, hierarchy = cv2.findContours(
  33          mask.astype(np.uint8), cv2.RETR_CCOMP, cv2.CHAIN_APPROX_SIMPLE
  34      )
  35      height = mask.shape[0]
  36      to_mm = lambda c: [((x + 0.5) * mm_per_pixel, (height - y - 0.5) * mm_per_pixel) for x, y in c[:, 0]]
  37      shells = []
  38      for index, contour in enumerate(contours):
  39          if hierarchy[0][index][3] != -1 or len(contour) < 3:
  40              continue
  41          holes = []
  42          child = hierarchy[0][index][2]
  43          while child != -1:
  44              if len(contours[child]) >= 3:
  45                  holes.append(to_mm(contours[child]))
  46              child = hierarchy[0][child][0]
  47          shells.append(Polygon(to_mm(contour), holes))
  48      half = 0.5 * mm_per_pixel
  49      geometry = unary_union([s.buffer(0) for s in shells]).buffer(half, join_style=2, mitre_limit=2.0)
  50      return geometry.simplify(simplify_mm, preserve_topology=True).buffer(0)

שלב א: מעקב קווי מתאר

RETR_CCOMP מחזיר היררכיה בשתי רמות: קווי מתאר חיצוניים (ההורה הוא 1−) וחורים, שהם הילדים שלהם, ונעבור עליהם בעזרת hierarchy[...][2] (ילד ראשון) ו־[0] (אח הבא). אי שבתוך חור מופיע שוב כקו מתאר חיצוני ברמה העליונה. CHAIN_APPROX_SIMPLE משאיר רק את קצוות הקטעים הישרים (אופקיים, אנכיים ואלכסוניים). OpenCV עוקב אחרי מרכזי פיקסלי הגבול של החומר, גם בחורים.

שלב ג: הרחבה בחצי פיקסל

DERIVED הפוליגון עובר דרך מרכזי פיקסלי הגבול, ולכן הוא קטן בחצי פיקסל מהמסכה האמיתית בכל צד. buffer(s/2, join_style=2, mitre_limit=2.0) מזיז כל צלע החוצה ב־s/2 עם פינות חדות (mitre). בצלעות אופקיות ואנכיות, הגבול עובר בדיוק לקצה הפיקסל. בחור, הקו עובר דרך מרכזי החומר שסביבו, וההרחבה מזיזה אותו לתוך החור, שוב בדיוק לקצה. בצלעות אלכסוניות של CHAIN_APPROX_SIMPLE התוצאה היא קו ישר ולא מדרגות, ולכן יש סטייה תת־פיקסלית.

def raster_ss(geom):                          # 4x supersampled, 1/16 sub-pixel precision
    img=np.zeros((N*S,N*S),np.uint8)
    f=lambda cs: np.round(np.array([(x/mm*S,(N-y/mm)*S) for x,y in cs])*16-8).astype(np.int32)
