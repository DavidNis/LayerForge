N=2304; mm=225/N; S=4
def raster_ss(geom):                          # 4x supersampled, 1/16 sub-pixel precision
    img=np.zeros((N*S,N*S),np.uint8)
    f=lambda cs: np.round(np.array([(x/mm*S,(N-y/mm)*S) for x,y in cs])*16-8).astype(np.int32)
    for poly in polygons(geom):
        cv2.fillPoly(img,[f(poly.exterior.coords)],1,shift=4)
        for r in poly.interiors: cv2.fillPoly(img,[f(r.coords)],0,shift=4)
    return img.reshape(N,S,N,S).sum((1,3))>=(S*S)/2
